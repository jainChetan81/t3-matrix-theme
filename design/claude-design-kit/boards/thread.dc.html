export function useProjectSearch(query: string) {
  const debounced = useDebouncedValue(query, 250);
  return useQuery({
    queryKey: ['projects', debounced],
    queryFn: ({ signal }) =>
      searchProjects(debounced, { signal }),
    enabled: debounced.length > 1,
  });
